RESULTS: F7 pilot of missingness-model forms (laptop, standalone, 2026-10-09)
Pilot only: 8 replications at n = 1e5, 300 at n = 500. se column is the standard error of the mean bias.
mnom = multinomial pattern model; in the sim mechanism it errors (two patterns only, where it equals joint) so its rows are NaN.

===== SCRIPT =====
# F7 pilot (laptop, pilot only): bias in b of five missingness-model forms under
# three mechanisms, as (weighted complete-case b) - (full-data b) per replication.
# (Z confounds M and Y and is omitted from the outcome model, so the population b
# is about 0.6, not 0.3; the full-data estimate is the benchmark.) Unstabilized weights, no trimming.
suppressMessages(library(nnet))
al_for <- function(target) uniroot(function(a) mean(plogis(a + rnorm(2e5))) - target, c(-3, 3))$root
set.seed(1); al <- al_for(.6)
gen <- function(n, mech) {
  C <- rnorm(n); Z <- rnorm(n); X <- rbinom(n, 1, plogis(.3 * C))
  M <- .3 * X + .5 * C + .7 * Z + rnorm(n)
  Y <- .3 * M + .2 * X + .5 * C + .7 * Z + rnorm(n)
  pr <- plogis(al + Z)
  if (mech == "ind")  { rm <- rbinom(n, 1, pr); ry <- rbinom(n, 1, pr) }
  if (mech == "sim")  { rm <- ry <- rbinom(n, 1, pr) }
  if (mech == "mono") { rm <- rbinom(n, 1, pr); ry <- rm * rbinom(n, 1, plogis(al + .8 + Z)) }
  data.frame(X, C, Z, M = ifelse(rm == 1, M, NA), Y = ifelse(ry == 1, Y, NA), rm, ry, M0 = M, Y0 = Y)
}
pcc <- function(d, form) {
  cc <- d$rm == 1 & d$ry == 1
  f <- function(y, dat, rhs) glm(reformulate(rhs, y), binomial, dat)
  switch(form,
    joint = fitted(f("cc", transform(d, cc = as.integer(cc)), c("X", "C", "Z"))),
    spv   = fitted(f("rm", d, c("X", "C", "Z"))) * fitted(f("ry", d, c("X", "C", "Z"))),
    seq   = { pm <- fitted(f("rm", d, c("X", "C", "Z")))
              m2 <- suppressWarnings(f("ry", d[d$rm == 1, ], c("X", "C", "Z")))
              pm * predict(m2, d, type = "response") },
    seqrev = { py <- fitted(f("ry", d, c("X", "C", "Z")))   # reverse order: Y first, then M | Y observed
              m2 <- suppressWarnings(f("rm", d[d$ry == 1, ], c("X", "C", "Z")))
              py * predict(m2, d, type = "response") },
    mnom  = { pat <- factor(2 * d$rm + d$ry)   # 3 = both observed
              m <- multinom(pat ~ X + C + Z, d, trace = FALSE)
              pp <- fitted(m)
              if (is.matrix(pp)) pp[, "3"] else pp })   # two levels: P(second level = "3")
}
est <- function(d, p) {
  cc <- d$rm == 1 & d$ry == 1; w <- 1 / p; dd <- d[cc, ]; w <- w[cc]
  unname(coef(lm(Y ~ X + M + C, dd, weights = w))["M"])
}
run <- function(n, reps) {
  out <- NULL
  for (mech in c("ind", "sim", "mono")) for (form in c("joint", "spv", "seq", "seqrev", "mnom")) {
    b <- replicate(reps, { d <- gen(n, mech); bf <- unname(coef(lm(Y0 ~ X + M0 + C, d))["M0"])
      tryCatch(est(d, pcc(d, form)) - bf, error = function(e) NA_real_) })
    out <- rbind(out, data.frame(n = n, mech, form, bias = mean(b, na.rm = TRUE), se = sd(b, na.rm = TRUE) / sqrt(sum(!is.na(b))), na = sum(is.na(b))))
  }
  out
}
set.seed(2); big <- run(1e5, 8)
set.seed(3); small <- run(500, 300)
print(rbind(big, small), digits = 3, row.names = FALSE)
